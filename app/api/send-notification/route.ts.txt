import { adminMessaging } from "@/lib/firebaseAdmin"; // استيراد الـ Firebase Admin
import { createClient } from "@supabase/supabase-js";

export async function POST(req: Request) {
  try {
    const { token } = await req.json();

    if (!token) {
      return Response.json({ error: "Token is required" }, { status: 400 });
    }

    // 1. حفظ التوكن في جدول Supabase (الطريقة الاعتيادية التي تستخدمها)
    // ... كود الحفظ في Supabase هنا ...

    // 2. تسجيل التوكن تلقائياً في الـ Topic (مثلاً اسم الـ Topic هو "all")
    const topic = "all";
    await adminMessaging.subscribeToTopic([token], topic);

    return Response.json({ success: true, message: "Subscribed to topic successfully" });
  } catch (error) {
    console.error("Error subscribing to topic:", error);
    return Response.json({ error: "Internal Server Error" }, { status: 500 });
  }
}
