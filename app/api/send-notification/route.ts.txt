import { adminMessaging } from "@/lib/firebaseAdmin";
import { createClient } from "@supabase/supabase-js";

// تهيئة عميل Supabase للباك إند
const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY! // مفتاح الـ Service Role لصلاحيات القراءة
);

export async function POST(req: Request) {
  try {
    const { title, body } = await req.json();

    // 1. جلب كل التوكنز المخزنة في جدول Supabase
    const { data: subscriptions, error } = await supabase
      .from("push_subscriptions")
      .select("token");

    if (error || !subscriptions || subscriptions.length === 0) {
      return Response.json({ error: "No subscribers found" }, { status: 404 });
    }

    // استخراج التوكنز في مصفوفة
    const tokens = subscriptions.map((sub) => sub.token).filter(Boolean);

    if (tokens.length === 0) {
      return Response.json({ error: "No valid tokens found" }, { status: 400 });
    }

    // 2. إرسال الإشعار دفعة واحدة لجميع التوكنز عبر Firebase Admin
    const message = {
      notification: {
        title: title || "إشعار جديد",
        body: body || "لديك إشعار جديد من تطبيق التأمين الوطني",
      },
      tokens: tokens, // مصفوفة التوكنز من Supabase
    };

    const response = await adminMessaging.sendEachForMulticast(message);

    return Response.json({
      success: true,
      successCount: response.successCount,
      failureCount: response.failureCount,
    });
  } catch (error) {
    console.error("Error sending broadcast notification:", error);
    return Response.json({ error: "Internal Server Error" }, { status: 500 });
  }
}
